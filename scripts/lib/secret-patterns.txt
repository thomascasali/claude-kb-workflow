# scripts/lib/secret-patterns.txt — regole per scripts/lib/secret-scan.sh
#
# Formato: NOME<TAB>REGEX_ERE<TAB>FLAG (FLAG opzionale)
#   Il separatore è un carattere TAB (non spazi, non "|": le regex usano
#   "|" per l'alternanza).
#
#   NOME   Identificatore stabile: appare nell'output come "file:riga:NOME".
#          Rinominare una regola invalida le voci esistenti in .secret-allowlist.
#   REGEX  Extended Regular Expression (grep -E), case-sensitive salvo flag CI.
#   FLAG   Opzionale, separati da virgola:
#            PLACEHOLDER    scarta la riga se il "valore" (token subito dopo
#                           := del match) sembra un segnaposto: ${VAR}, <...>,
#                           xxx, changeme, your_..., example, TODO, null,
#                           vuoto, "...", oppure è più corto di 8 caratteri
#                           o inizia con un carattere che un secret vero non
#                           ha in testa (` * ( $ < % {).
#            FMTPLACEHOLDER per le regole di FORMATO (non chiave=valore):
#                           scarta il match se la riga contiene 6+ X/x
#                           consecutive, oppure "{" o "<" (template).
#            CI             regola case-insensitive.
#
# Righe vuote e righe che iniziano per # sono ignorate. Non scrivere mai qui
# un secret come esempio: le regex sono già abbastanza specifiche.
#
# Confine iniziale "(^|[^A-Za-z0-9_-])": i prefissi corti (sk-, re_, AIza...)
# altrimenti matchano come sottostringa di identificatori ordinari
# ("re_" dentro "measure_attention"). Un secret vero è preceduto da inizio
# riga, spazio, virgolette, "=" o ":", mai da un'altra lettera.

PRIVATE_KEY	-----BEGIN (RSA |EC |OPENSSH |DSA |PGP )?PRIVATE KEY-----
OPENAI_KEY	(^|[^A-Za-z0-9_-])sk-[A-Za-z0-9_-]{16,}
ANTHROPIC_KEY	(^|[^A-Za-z0-9_-])sk-ant-[A-Za-z0-9_-]{16,}
STRIPE_LIVE_SECRET_KEY	(^|[^A-Za-z0-9_-])sk_live_[A-Za-z0-9]{16,}
STRIPE_LIVE_RESTRICTED_KEY	(^|[^A-Za-z0-9_-])rk_live_[A-Za-z0-9]{16,}
GITHUB_TOKEN	(^|[^A-Za-z0-9_-])gh[pousr]_[A-Za-z0-9]{20,}
GITHUB_FINE_GRAINED_TOKEN	(^|[^A-Za-z0-9_-])github_pat_[A-Za-z0-9_]{20,}
AWS_ACCESS_KEY_ID	(^|[^A-Za-z0-9_-])AKIA[0-9A-Z]{16}
GOOGLE_API_KEY	(^|[^A-Za-z0-9_-])AIza[0-9A-Za-z_-]{35}	FMTPLACEHOLDER
SLACK_TOKEN	(^|[^A-Za-z0-9_-])xox[bpoa]-[A-Za-z0-9-]{10,}
TELEGRAM_BOT_TOKEN	(^|[^0-9])[0-9]{8,10}:[A-Za-z0-9_-]{35}
RESEND_API_KEY	(^|[^A-Za-z0-9_-])re_[A-Za-z0-9]{20,}
JWT	(^|[^A-Za-z0-9_-])eyJ[A-Za-z0-9_-]{10,}\.eyJ[A-Za-z0-9_-]{10,}\.[A-Za-z0-9_-]{10,}
URL_WITH_CREDENTIALS	:\/\/[A-Za-z0-9._%+-]+:[^/@[:space:]]+@	FMTPLACEHOLDER
#
# SECRET_ASSIGNMENT: CASE-SENSITIVE. Il nome della chiave è riconosciuto in
# due sole forme, per evitare falsi positivi sulla prosa:
#   1. stile env, tutto MAIUSCOLO, ovunque nella riga:     DB_PASSWORD=...
#   2. minuscolo, SOLO tra virgolette (chiave JSON) o a inizio riga (YAML).
# Il "valore" è un token singolo subito dopo := ; il filtro PLACEHOLDER
# scarta poi valori corti e segnaposto noti. La parola chiave deve essere un
# segmento intero delimitato da "_" (TOKEN non scatta dentro DESIGN_TOKENS).
SECRET_ASSIGNMENT	(^|[^A-Za-z0-9])([A-Z0-9]+_)*(PASSWORD|PASSWD|SECRET|TOKEN|API_?KEY|PRIVATE_KEY)(_[A-Z0-9]+)*[[:space:]]*[:=][[:space:]]*[^[:space:]]|"([a-z0-9]+_)*(password|passwd|secret|token|api_?key|private_key)(_[a-z0-9]+)*"[[:space:]]*:[[:space:]]*[^[:space:]]|^[[:space:]]*([a-z0-9]+_)*(password|passwd|secret|token|api_?key|private_key)(_[a-z0-9]+)*:[[:space:]]+[^[:space:]]	PLACEHOLDER
#
# SECRET_ASSIGNMENT_PROSE: variante volutamente larga (case-insensitive,
# parola chiave + connettore corto + :/=/è/is), pensata per le note di
# memoria scritte in prosa ("- Password: ...", "il token e' ..."). Attiva
# SOLO dove secret-scope.txt la include (cartelle di memoria): altrove è
# disattivata, o la prosa ordinaria la farebbe scattare in massa.
SECRET_ASSIGNMENT_PROSE	(password|passwd|pwd|secret|token|api[ _-]?key|client[ _-]?secret|private[ _-]?key)[A-Za-z0-9 '-]{0,20}(:|=|[[:space:]](è|È)[[:space:]]|[[:space:]]e'[[:space:]]|[[:space:]]is[[:space:]])[[:space:]]*[^[:space:]]	PLACEHOLDER,CI
